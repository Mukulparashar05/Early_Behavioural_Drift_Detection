NUMERIC = ['Air temperature [K]', 'Process temperature [K]', 'Rotational speed [rpm]', 'Torque [Nm]', 'Tool wear [min]']

# Phase 1: reproducibly split records while preserving the observed failure ratio.
def stratified_split(y, seed=42):
    rng=np.random.default_rng(seed); splits=[[],[],[]]
    for label in np.unique(y):
        ix=np.flatnonzero(y==label); rng.shuffle(ix)
        n=len(ix); a=int(.70*n); b=int(.85*n)
        splits[0].extend(ix[:a]); splits[1].extend(ix[a:b]); splits[2].extend(ix[b:])
    return tuple(np.array(sorted(s),dtype=int) for s in splits)

# Phase 2: fit category levels and sensor scaling from training rows only.
def prepare(df, train_idx):
    """Fit category vocabulary and standardization on training rows only."""
    cats=sorted(df['Type'].dropna().astype(str).unique().tolist())
    means=df.iloc[train_idx][NUMERIC].mean().to_numpy(float)
    std=df.iloc[train_idx][NUMERIC].std(ddof=0).replace(0,1).to_numpy(float)
    def transform(frame):
        n=(frame[NUMERIC].to_numpy(float)-means)/std
        c=frame['Type'].astype(str).to_numpy()
        oh=np.column_stack([c==v for v in cats]).astype(float)
        return np.column_stack([n,oh])
    return transform, {'numeric_features':NUMERIC,'category_levels':cats,'mean':means.tolist(),'scale':std.tolist(),'input_features':NUMERIC+[f'Type={c}' for c in cats]}

def reconstruction_scores(model, x):
    return np.mean((model.forward(x)-x)**2,axis=1)

def percentile_threshold(normal_scores, percentile=99):
    a=np.asarray(normal_scores,float)
    if len(a)==0 or not np.isfinite(a).all(): raise ValueError('Need finite normal validation scores')
    return float(np.quantile(a,percentile/100))

# Phase 3: a compact neural autoencoder implemented with NumPy and Adam updates.
class Autoencoder:
    """Small one-hidden-layer tanh bottleneck autoencoder trained by Adam."""
    def __init__(self,n_in,latent,seed=42,lr=.01):
        rng=np.random.default_rng(seed); self.lr=lr
        self.W1=rng.normal(0,math.sqrt(2/(n_in+latent)),(n_in,latent)); self.b1=np.zeros(latent)
        self.W2=rng.normal(0,math.sqrt(2/(n_in+latent)),(latent,n_in)); self.b2=np.zeros(n_in)
        self.params=[self.W1,self.b1,self.W2,self.b2]; self.m=[np.zeros_like(p) for p in self.params]; self.v=[np.zeros_like(p) for p in self.params]; self.t=0
    def forward(self,x): return np.tanh(x@self.W1+self.b1)@self.W2+self.b2
    def train_batch(self,x):
        z=np.tanh(x@self.W1+self.b1); y=z@self.W2+self.b2; n=max(1,x.shape[0]); dy=2*(y-x)/(x.shape[1]*n)
        grads=[x.T@( (dy@self.W2.T)*(1-z*z)), np.sum((dy@self.W2.T)*(1-z*z),axis=0), z.T@dy, dy.sum(axis=0)]
        self.t+=1
        for i,(p,g) in enumerate(zip(self.params,grads)):
            self.m[i]=.9*self.m[i]+.1*g; self.v[i]=.999*self.v[i]+.001*g*g
            mh=self.m[i]/(1-.9**self.t); vh=self.v[i]/(1-.999**self.t)
            p-=self.lr*mh/(np.sqrt(vh)+1e-8)
    def save(self,path):
        with open(path,'wb') as f: pickle.dump({'W1':self.W1,'b1':self.b1,'W2':self.W2,'b2':self.b2},f)

# Phase 4: calculate imbalance-aware test metrics and ranked average precision.
def metrics(y, pred, scores):
    y=np.asarray(y,int); pred=np.asarray(pred,int); tp=int(((y==1)&(pred==1)).sum()); fp=int(((y==0)&(pred==1)).sum()); fn=int(((y==1)&(pred==0)).sum()); tn=int(((y==0)&(pred==0)).sum())
    p=tp/(tp+fp) if tp+fp else 0.; r=tp/(tp+fn) if tp+fn else 0.; f=2*p*r/(p+r) if p+r else 0.
    # Average precision using ranked positive examples; robust to ties.
    order=np.argsort(-np.asarray(scores)); ys=y[order]; positives=int(y.sum()); ap=0.
    if positives:
        cumulative=np.cumsum(ys); ranks=np.arange(1,len(y)+1); ap=float(np.sum((ys==1)*cumulative/ranks)/positives)
    return {'precision':p,'recall':r,'f1':f,'pr_auc_average_precision':ap if positives else None,'tn':tn,'fp':fp,'fn':fn,'tp':tp,'false_positive_rate':fp/(fp+tn) if fp+tn else 0.,'n_positive':int(y.sum()),'n_negative':int((y==0).sum())}

# Phase 5: aggregate threshold exceedances over adjacent rows; these are not timestamps.
def drift_analysis(scores, threshold, window=25, fraction=.6):
    scores=np.asarray(scores,float); elevated=(scores>threshold).astype(float); roll=np.convolve(elevated,np.ones(window)/window,mode='full')[:len(scores)]
    roll[:min(window-1,len(scores))]=np.nan
    drift=(roll>=fraction); return roll,drift

# Phase 6: apply transparent heuristic warning states; uncertainty is not probability.
def classify_warning(score, threshold, drift_score, uncertainty, margin=.1):
    if not np.isfinite(drift_score): return 'UNCERTAIN','low','Insufficient observations to complete rolling window'
    if abs(score/threshold-1)<=margin: return 'UNCERTAIN','moderate','Anomaly score is close to validation threshold'
    if drift_score>=.8: return 'SUSTAINED_HIGH','high','At least 80% of the rolling window exceeds threshold'
    if drift_score>=.6: return 'POTENTIAL_DRIFT','moderate','At least 60% of the rolling window exceeds threshold'
    return ('POTENTIAL_DRIFT','low','Anomaly score exceeds threshold') if score>threshold else ('NORMAL','low','Score below threshold; persistence rule not met')

def _expected_path(n):
    if n<=1: return 0.
    return 2*math.log(n-1)+0.5772156649-2*(n-1)/n

class IsolationForest:
    """Small axis-aligned isolation forest implemented with NumPy/Python."""
    def __init__(self, n_trees=100, sample_size=256, seed=42):
        self.n_trees=n_trees; self.sample_size=sample_size; self.rng=np.random.default_rng(seed); self.trees=[]
    def _build(self,x,depth,max_depth):
        if depth>=max_depth or len(x)<=1: return (len(x),)
        candidates=np.flatnonzero(np.ptp(x,axis=0)>1e-12)
        if len(candidates)==0: return (len(x),)
        f=int(self.rng.choice(candidates)); lo=float(x[:,f].min()); hi=float(x[:,f].max()); cut=float(self.rng.uniform(lo,hi))
        left=x[:,f]<cut; nleft=int(left.sum())
        if nleft==0 or nleft==len(x): return (len(x),)
        return (f,cut,self._build(x[left],depth+1,max_depth),self._build(x[~left],depth+1,max_depth))
    def fit(self,x):
        x=np.asarray(x,float); n=min(self.sample_size,len(x)); md=int(math.ceil(math.log2(max(n,2))))
        self.trees=[self._build(x[ix],0,md) for ix in (self.rng.choice(len(x),n,replace=False) for _ in range(self.n_trees))]
        return self
    def _path(self,node,row,depth=0):
        if len(node)==1: return depth+_expected_path(node[0])
        f,cut,left,right=node
        return self._path(left if row[f]<cut else right,row,depth+1)
    def score_samples(self,x):
        x=np.asarray(x,float); c=_expected_path(min(self.sample_size,256)) or 1.
        return np.array([2**(-np.mean([self._path(tree,row) for tree in self.trees])/c) for row in x])

def save_plot(path,title,series,labels,colors=None):
    """Compact publication-style bar/line plot using Pillow (no plotting stack required)."""
    w,h=1100,620; im=Image.new('RGB',(w,h),'white'); d=ImageDraw.Draw(im); left,top,right,bottom=95,80,1050,520
    d.text((left,25),title,fill='#162536'); d.line((left,top,left,bottom),fill='#435466',width=2); d.line((left,bottom,right,bottom),fill='#435466',width=2)
    vals=np.asarray(series,dtype=float); vals=vals[np.isfinite(vals)]
    if len(vals)==0: vals=np.array([0.,1.])
    lo=float(vals.min()); hi=float(vals.max()); span=(hi-lo) or 1.
    for k in range(5):
        yy=top+(bottom-top)*k/4; d.line((left,yy,right,yy),fill='#e5eaf0'); d.text((18,yy-7),f'{hi-span*k/4:.3g}',fill='#34495e')
    if len(series)>1:
        points=[]; arr=np.asarray(series,float); n=len(arr)
        for i,v in enumerate(arr):
            if np.isfinite(v): points.append((int(left+(right-left)*i/max(n-1,1)),int(bottom-(v-lo)/span*(bottom-top))))
        for a,b in zip(points,points[1:]): d.line((*a,*b),fill=(colors or ['#1677aa'])[0],width=2)
    else:
        for i,v in enumerate(series):
            x=left+100+i*220; y=bottom-(float(v)-lo)/span*(bottom-top); d.rectangle((x,y,x+90,bottom),fill=(colors or ['#1677aa'])[i%len(colors or ['#1677aa'])]); d.text((x,bottom+10),labels[i] if i<len(labels) else str(i),fill='#34495e')
    d.text((left,bottom+45),labels[0] if len(series)>1 and labels else '',fill='#34495e'); im.save(path)

# Main pipeline: audit -> train-only preprocessing -> normal-only fit -> test evaluation.
def run(csv_path, config_path=None):
    cfg=json.loads(Path(config_path).read_text()) if config_path else json.loads((ROOT/'config.json').read_text())
    global SEED
    SEED=cfg['seed']; np.random.seed(SEED)
    dirs=[OUT/'figures',OUT/'tables',OUT/'models',OUT/'metrics',ROOT/'reports']
    for d in dirs: d.mkdir(parents=True,exist_ok=True)
    df=pd.read_csv(csv_path)
    required=set(['UDI','Product ID','Type',*NUMERIC,*TARGETS]); missing=required-set(df.columns)
    if missing: raise ValueError(f'Dataset schema missing columns: {sorted(missing)}')
    y=df['Machine failure'].astype(int).to_numpy(); train_idx,val_idx,test_idx=stratified_split(y,SEED)
    transform,schema=prepare(df,train_idx); x=transform(df)
    audit={'rows':len(df),'columns':len(df.columns),'column_names':df.columns.tolist(),'dtypes':{k:str(v) for k,v in df.dtypes.items()},'missing':df.isna().sum().astype(int).to_dict(),'duplicate_rows':int(df.duplicated().sum()),'descriptive_statistics':df.describe(include='all').fillna('').to_dict(),'numeric_ranges':{c:[float(df[c].min()),float(df[c].max())] for c in NUMERIC},'categorical_distributions':{c:df[c].value_counts(dropna=False).to_dict() for c in ['Type']},'failure_balance':df['Machine failure'].value_counts().sort_index().to_dict(),'failure_modes':{c:int(df[c].sum()) for c in TARGETS[1:]},'feature_correlations':df[NUMERIC].corr().round(6).to_dict(),'identifier_columns':['UDI','Product ID'],'sensor_variables':NUMERIC,'categorical_columns':['Type'],'targets_excluded_from_inputs':TARGETS}
    (OUT/'metrics'/'data_audit.json').write_text(json.dumps(audit,indent=2,default=str)); df[NUMERIC+TARGETS].describe().to_csv(OUT/'tables'/'descriptive_statistics.csv')
    trn=train_idx[y[train_idx]==0]; valn=val_idx[y[val_idx]==0]
    if min(len(trn),len(valn))<2: raise ValueError('Insufficient normal train/validation observations')
    model=Autoencoder(x.shape[1],cfg['latent_dim'],SEED,cfg['learning_rate']); rng=np.random.default_rng(SEED); best=None; bestval=float('inf'); patience=0; history=[]
    for epoch in range(cfg['epochs']):
        order=rng.permutation(trn); model.train_batch(x[order[:len(order)]]) if False else None
        for start in range(0,len(order),cfg['batch_size']): model.train_batch(x[order[start:start+cfg['batch_size']]])
        tl=float(np.mean((model.forward(x[trn])-x[trn])**2)); vl=float(np.mean((model.forward(x[valn])-x[valn])**2)); history.append({'epoch':epoch+1,'train_loss':tl,'validation_loss':vl})
        if vl<bestval-1e-7: bestval=vl; best=[p.copy() for p in model.params]; patience=0
        else: patience+=1
        if patience>=cfg['patience']: break
    model.W1,model.b1,model.W2,model.b2=best
    model.save(OUT/'models'/'autoencoder_numpy.pkl'); (OUT/'models'/'preprocessing_schema.json').write_text(json.dumps(schema,indent=2)); pd.DataFrame(history).to_csv(OUT/'tables'/'training_history.csv',index=False)
    threshold=percentile_threshold(reconstruction_scores(model,x[valn]),cfg['threshold_percentile'])
    # Held-out scoring; labels are joined only after scores and flags are computed.
    test_scores=reconstruction_scores(model,x[test_idx]); test_pred=(test_scores>threshold).astype(int); test_y=y[test_idx]
    statmean=x[trn].mean(axis=0); statstd=x[trn].std(axis=0)+1e-8; stat_score=np.mean(((x[test_idx]-statmean)/statstd)**2,axis=1); stat_thr=percentile_threshold(np.mean(((x[valn]-statmean)/statstd)**2,axis=1),cfg['threshold_percentile'])
    forest=IsolationForest(n_trees=100,sample_size=256,seed=SEED).fit(x[trn]); if_val=forest.score_samples(x[valn]); if_test=forest.score_samples(x[test_idx]); if_thr=percentile_threshold(if_val,cfg['threshold_percentile'])
    metric={'autoencoder':metrics(test_y,test_pred,test_scores),'statistical_zscore':metrics(test_y,(stat_score>stat_thr).astype(int),stat_score),'isolation_forest':metrics(test_y,(if_test>if_thr).astype(int),if_test),'threshold':threshold,'threshold_rule':f'{cfg["threshold_percentile"]}th percentile of normal-only validation reconstruction errors','isolation_forest_threshold':if_thr,'isolation_forest_threshold_rule':f'{cfg["threshold_percentile"]}th percentile of normal-only validation Isolation Forest scores','split_counts':{'train':len(train_idx),'validation':len(val_idx),'test':len(test_idx),'normal_train':len(trn),'normal_validation':len(valn)},'feature_schema':schema,'configuration':cfg,'versions':{'python':platform.python_version(),'numpy':np.__version__,'pandas':pd.__version__},'limitations':['AI4I is synthetic and the supplied CSV has no timestamp or continuous per-machine identity.','Temporal experiments use source row order and are not real chronological early warnings.','The final test partition is stratified and held out; label is used only for normal-train construction, split stratification and post-score evaluation.','Thresholds are selected from normal validation data.','Average precision is reported as the PR-AUC summary.']}
    # temporal prototype is explicitly ordered-row analysis
    all_scores=reconstruction_scores(model,x); rolling,drift=drift_analysis(all_scores,threshold,cfg['rolling_window'],cfg['drift_fraction'])
    # empirical bootstrap reference uncertainty from normal validation scores
    ref=reconstruction_scores(model,x[valn]); boots=np.random.default_rng(SEED).choice(ref,size=(400,len(ref)),replace=True).mean(axis=1); uncertainty=float(np.std(boots))
    outputs=[]
    for pos,(idx,score) in enumerate(zip(range(len(df)),all_scores)):
        ds=float(rolling[pos]) if np.isfinite(rolling[pos]) else float('nan'); state,severity,reason=classify_warning(float(score),threshold,ds,uncertainty,cfg['warning_uncertainty_margin'])
        row={'source_row_index':int(idx),'UDI':int(df.iloc[idx]['UDI']),'Product ID':df.iloc[idx]['Product ID'],'reconstruction_error':float(score),'threshold':threshold,'anomaly':int(score>threshold),'ground_truth_failure_evaluation_only':int(y[idx]),'rolling_exceedance_fraction':ds,'drift_detected':bool(drift[pos]) if np.isfinite(ds) else False,'warning_state':state,'severity':severity,'uncertainty_bootstrap_sd':uncertainty,'warning_reason':reason}
        row.update({c:float(df.iloc[idx][c]) for c in NUMERIC}); outputs.append(row)
    pd.DataFrame(outputs).to_csv(OUT/'tables'/'observation_scores.csv',index=False)
    pd.DataFrame([metric['autoencoder'],metric['statistical_zscore'],metric['isolation_forest']],index=['autoencoder','statistical_zscore','isolation_forest']).to_csv(OUT/'tables'/'test_metrics.csv')
    (OUT/'metrics'/'results.json').write_text(json.dumps(metric,indent=2))
    save_lines(OUT/'figures'/'loss_curve.png','Autoencoder training and validation loss',[r['train_loss'] for r in history],[r['validation_loss'] for r in history])
    # score distributions displayed as reproducible histograms rendered with Pillow.
    save_hist(OUT/'figures'/'score_distributions.png',test_scores[test_y==0],test_scores[test_y==1],threshold)
    save_plot(OUT/'figures'/'ordered_row_drift.png','Ordered-row experiment: anomaly score (source order)',all_scores[:800],['Source row order'],['#1677aa'])
    save_bars(OUT/'figures'/'target_distribution.png','Machine failure class balance',[int((y==0).sum()),int((y==1).sum())],['No failure','Failure'])
    save_corr(OUT/'figures'/'correlation_heatmap.png',df[NUMERIC].corr().to_numpy(),NUMERIC)
    save_confusion(OUT/'figures'/'confusion_matrix.png',metric['autoencoder'])
    save_precision_recall(OUT/'figures'/'precision_recall_curve.png',test_y,test_scores)
    save_comparison(OUT/'figures'/'baseline_comparison.png',metric)
    save_sensor_distributions(OUT/'figures'/'sensor_distributions.png',df)
    synthetic_result=make_synthetic(OUT/'figures'/'synthetic_drift_demo.png',model,x[trn],threshold,cfg)
    metric['synthetic_drift_experiment']=synthetic_result
    (OUT/'metrics'/'results.json').write_text(json.dumps(metric,indent=2))
    write_report(metric,history,threshold)
    print(f'Loaded {len(df)} rows; train/validation/test={len(train_idx)}/{len(val_idx)}/{len(test_idx)}; normal train={len(trn)}')
    print(f'Features={x.shape[1]}; epochs={len(history)}; threshold={threshold:.8g}')
    print('Held-out autoencoder metrics:',json.dumps(metric['autoencoder'],sort_keys=True))
    print('Outputs written under',OUT)

def make_synthetic(path,model,normal_x,threshold,cfg):
    rng=np.random.default_rng(cfg['seed']+10); n=600; change=n//2
    seq=normal_x[rng.integers(0,len(normal_x),size=n)].copy(); seq[change:,0]+=3.0; scores=reconstruction_scores(model,seq)
    rolling,drift=drift_analysis(scores,threshold,cfg['rolling_window'],cfg['drift_fraction'])
    detected=np.flatnonzero(drift[change:])+change; first=int(detected[0]) if len(detected) else None
    # Show anomaly score, threshold, injected shift and persistence alarm.
    w,h=1100,620; im=Image.new('RGB',(w,h),'white'); d=ImageDraw.Draw(im); l,t,r,b=95,80,1050,510

    pts=[(int(l+(r-l)*i/(n-1)),int(b-(v-lo)/span*(b-t))) for i,v in enumerate(scores)]
    for p,q in zip(pts,pts[1:]): d.line((*p,*q),fill='#1677aa',width=2)
    tx=int(l+(r-l)*change/(n-1)); ty=int(b-(threshold-lo)/span*(b-t)); d.line((tx,t,tx,b),fill='#cf725f',width=2); d.line((l,ty,r,ty),fill='#222222',width=2)
